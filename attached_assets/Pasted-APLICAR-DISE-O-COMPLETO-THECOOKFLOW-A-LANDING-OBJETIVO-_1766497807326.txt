PASO 1: COPIAR ASSETS DESDE REPO PÚBLICO

Ejecutar en Shell:

# Limpiar
rm -rf temp-brand

# Crear estructura
mkdir -p public/bg
mkdir -p public/logos
mkdir -p public/icons

# Clonar repo público
git clone https://github.com/RUPERDFN/tcf-brand-master.git temp-brand

# Copiar fondos
cp temp-brand/fondos/bg.png public/bg/
cp temp-brand/fondos/textura-grunge-oscura.jpg public/bg/

# Copiar logos
cp temp-brand/logos/logo.PNG public/logos/logo.png
cp temp-brand/logos/chef.png public/logos/
cp temp-brand/logos/menu.png public/logos/
cp temp-brand/logos/lista.png public/logos/
cp temp-brand/logos/recetas.png public/logos/

# Copiar iconos
cp temp-brand/logos/tomate.png public/icons/
cp temp-brand/logos/zanahoria.png public/icons/
cp temp-brand/logos/pimiento.png public/icons/
cp temp-brand/logos/limon.png public/icons/
cp temp-brand/logos/hojas.png public/icons/

# Limpiar
rm -rf temp-brand

# Verificar
ls -la public/bg/
ls -la public/logos/
ls -la public/icons/
